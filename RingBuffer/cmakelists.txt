cmake_minimum_required(VERSION 3.20)

include(FetchContent)

FetchContent_Declare(
    googletest
    GIT_REPOSITORY https://github.com/google/googletest.git
    GIT_TAG v1.17.0
)

FetchContent_MakeAvailable(googletest)

add_executable(test
    Test.cpp
)
add_library(ringbuff
    ringbuff.c 
)
target_include_directories(ringbuff
    PUBLIC
        ${CMAKE_CURRENT_SOURCE_DIR}
)
target_link_libraries(test
    PRIVATE
        ringbuff
        gtest_main
)
target_compile_options(ringbuff PRIVATE --coverage)
target_link_options(ringbuff PRIVATE --coverage)

target_link_options(test PRIVATE --coverage)
include(GoogleTest)
